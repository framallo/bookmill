% Centered, two-line chapter openers (illustrated children's books).
% Line 1: the localized label "Capítulo N" (es) / "Chapter N" (en) — via babel's
%   \chaptername, so it follows the document language automatically.
% Line 2: the chapter title, centered below.
% Unnumbered chapters (\chapter*, e.g. the glossary/guide) show only the
% centered title, with no label line.
\usepackage{titlesec}
\usepackage{xcolor}
% Warm terracotta for chapter openers (the interior prints in color).
\definecolor{islatitle}{HTML}{C2571C}

% Pandoc defaults to no section numbering (\setcounter{secnumdepth}{-\maxdimen}),
% which makes every \chapter unnumbered. We want NUMBERED chapters (so the
% opener can print "Capítulo N" / "Chapter N") but no section numbering, so set
% secnumdepth to 0: chapters get numbers, sections and deeper do not.
\setcounter{secnumdepth}{0}

% Numbered chapters: label line ("Capítulo \thechapter") + centered title below.
\titleformat{\chapter}[display]
  {\normalfont\centering\bfseries\color{islatitle}}
  {\Large\mdseries\scshape\chaptername\ \thechapter}
  {12pt}
  {\Huge}

% Unnumbered chapters (\chapter*): centered title only, no label line.
\titleformat{name=\chapter,numberless}[display]
  {\normalfont\centering\bfseries\color{islatitle}}
  {}
  {0pt}
  {\Huge}

% Sensible, warm vertical spacing around the opener.
\titlespacing*{\chapter}{0pt}{20pt}{40pt}
