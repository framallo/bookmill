% Facing-page illustration plates (illustrated children's books).
% \cleartoverso jumps to a left/verso page so the following \chapter (openright)
% lands on the facing recto — full-page plate on the left, chapter opener right.
\usepackage{graphicx}
\usepackage{eso-pic}
\newcommand{\cleartoverso}{\clearpage\ifodd\value{page}\hbox{}\thispagestyle{empty}\clearpage\fi}

% Full-bleed plate: fill the ENTIRE paper (paperwidth x paperheight) so the
% artwork reaches all four physical edges. KDP requires edge-to-edge images on a
% bleed interior to extend PAST the trim line into the 0.125in bleed; fitting the
% image inside the trim leaves a white border and is rejected as "insufficient
% bleed". The outer 0.125in is trimmed off by KDP, as intended. Render plates at
% (or slightly past) the bleed page aspect to avoid stretch. Starred form affects
% only this one page.
\newcommand{\AddThisPageImage}[1]{%
  \AddToShipoutPictureBG*{%
    \put(0,0){\includegraphics[width=\paperwidth,height=\paperheight]{#1}}%
  }%
}
