% Facing-page illustration plates (illustrated children's books).
% \cleartoverso jumps to a left/verso page so the following \chapter (openright)
% lands on the facing recto — full-page plate on the left, chapter opener right.
\usepackage{graphicx}
\usepackage{eso-pic}
\newcommand{\cleartoverso}{\clearpage\ifodd\value{page}\hbox{}\thispagestyle{empty}\clearpage\fi}

% Place an illustration as large as possible, centered, aspect preserved, but
% fitted to the TRIM box (0.125in inside each paper edge = the bleed amount) so
% content at the artwork's edge lands at the cut line and is NOT lost in the
% bleed when KDP trims. The starred form affects only this one page.
\newcommand{\AddThisPageImage}[1]{%
  \AddToShipoutPictureBG*{%
    \put(0,0){%
      \parbox[b][\paperheight][c]{\paperwidth}{%
        \centering
        \includegraphics[width=\dimexpr\paperwidth-0.25in\relax,height=\dimexpr\paperheight-0.25in\relax,keepaspectratio]{#1}%
      }%
    }%
  }%
}
